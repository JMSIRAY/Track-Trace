<script>
function executeTrackerSearch() {
    const inputVal = document.getElementById("branchSearchInput").value.trim();
    const resultBox = document.getElementById("trackerResultBox");
    
    if (!inputVal) return alert("Mangyaring maglagay ng kahit ilang letra ng Branch name.");
    
    // Tinatawag nito ang searchData function na nasa Code.gs
    google.script.run.withSuccessHandler(function(data) {
        if (data.found) {
            document.getElementById("displayBranch").innerText = data.branch;
            document.getElementById("displayItem").innerText = data.item;
            document.getElementById("displayRemarks").innerText = data.remarks ? data.remarks : "Walang opisyal na ulat.";
            
            const statusPill = document.getElementById("displayStatus");
            const remarksBorder = document.getElementById("remarksIndicator");
            statusPill.innerText = data.status;
            
            const lowerCaseStatus = data.status.toLowerCase();
            
            if (lowerCaseStatus.includes("delivered") || lowerCaseStatus.includes("kumpleto")) {
                statusPill.style.backgroundColor = "var(--success)";
                remarksBorder.style.borderLeftColor = "var(--success)";
            } else if (lowerCaseStatus.includes("delivery") || lowerCaseStatus.includes("biyahe") || lowerCaseStatus.includes("transit")) {
                statusPill.style.backgroundColor = "var(--warning)";
                remarksBorder.style.borderLeftColor = "var(--warning)";
            } else {
                statusPill.style.backgroundColor = "var(--neutral)";
                remarksBorder.style.borderLeftColor = "var(--rusi-black)";
            }
            
            resultBox.style.display = "block";
        } else {
            alert("Hindi mahanap ang branch. Pakisuri ang iyong tinype.");
            resultBox.style.display = "none";
        }
    }).searchData(inputVal);
}
</script>
